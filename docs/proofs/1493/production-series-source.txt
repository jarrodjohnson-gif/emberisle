#!/usr/bin/env bash
export PATH=/workspace/work/runtime/node_modules/node/bin:$PATH
series_dir=/workspace/work/resume-nov9/1493-production-series
mkdir -p "$series_dir"
git rev-parse HEAD > "$series_dir/source-head"
for source_file in scripts/connection-health-prove.mjs src/components/game/EmberisleApp.tsx src/components/game/chunks.ts; do
  printf '%s ' "$source_file" >> "$series_dir/source-blobs"
  git hash-object "$source_file" >> "$series_dir/source-blobs"
done
node --version > "$series_dir/node-version"
node --input-type=module -e 'setInterval(()=>{const until=performance.now()+100;while(performance.now()<until){}},200)' &
load_pid=$!
trap 'kill "$load_pid" 2>/dev/null || true' EXIT
for run in $(seq 1 20); do
  date -u +%FT%TZ > "$series_dir/$run.started"
  node scripts/connection-health-prove.mjs > "$series_dir/$run.log" 2>&1
  result=$?
  printf '%s\n' "$result" > "$series_dir/$run.exit"
  printf 'run %s exit %s\n' "$run" "$result"
  if [ "$result" -ne 0 ]; then tail -30 "$series_dir/$run.log"; exit "$result"; fi
done
