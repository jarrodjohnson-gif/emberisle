#!/bin/bash
set -u
cd /workspace/emberisle-adopt1527 || exit 2
out=docs/proofs/1525/adoption-2026-11-12/historical-runtime
historical_node=/workspace/emberisle-historical-runtime/node_modules/node/bin/node
export CHROMIUM_PATH=/workspace/emberisle-historical-runtime/browsers/chromium-1243/chrome-linux64/chrome
[ "$(git rev-parse HEAD)" = f9f7ea5fe8c18ad637fc4660bc258a6b626c98cd ] || exit 2
sha256sum -c "$out/frozen-source-sha256.txt" > "$out/start-hashes.log" || exit 2
git rev-parse HEAD > "$out/source-head.txt"
"$historical_node" --version > "$out/node-version.txt"
"$CHROMIUM_PATH" --version > "$out/browser-version.txt"
cat /sys/fs/cgroup/cpu.max > "$out/cpu-max.txt"
"$historical_node" --input-type=module -e 'import { chromium } from "./scripts/proof-browser.mjs"; const b=await chromium.launch({args:["--use-gl=angle","--use-angle=swiftshader","--enable-unsafe-swiftshader"]}); console.log(await b.version()); await b.close();' > "$out/preflight.log" 2>&1
setup_result=$?
printf '%s\n' "$setup_result" > "$out/preflight.exit.txt"
[ "$setup_result" -eq 0 ] || exit "$setup_result"
"$historical_node" --input-type=module -e 'setInterval(()=>{const until=performance.now()+100;while(performance.now()<until){}},200)' &
load_pid=$!
trap 'kill "$load_pid" 2>/dev/null || true' EXIT
ONLY=online "$historical_node" scripts/.sol1527-boundary-diagnostic.mjs > "$out/diagnostic.log" 2>&1
result=$?
printf '%s\n' "$result" > "$out/diagnostic.exit.txt"
sha256sum -c "$out/frozen-source-sha256.txt" > "$out/end-hashes.log" || exit 2
tail -n 8 "$out/diagnostic.log"
exit "$result"
