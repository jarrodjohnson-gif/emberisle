#!/usr/bin/env bash
set -u
export PATH=/workspace/work/runtime/node_modules/node/bin:$PATH
cd /workspace/work/1518 || exit 2
out=/workspace/work/resume-nov9/1518-followup-full
[ ! -e "$out.started" ] || exit 2
git rev-parse HEAD > "$out.source"
git rev-parse HEAD:scripts/turn-gain-prove.mjs > "$out.blob"
cat /sys/fs/cgroup/cpu.max > "$out.cpu"
date -u +%FT%TZ > "$out.started"
node --input-type=module -e 'setInterval(()=>{const until=performance.now()+100;while(performance.now()<until){}},200)' &
load_pid=$!
trap 'kill "$load_pid" 2>/dev/null || true' EXIT
node scripts/turn-gain-prove.mjs > "$out.log" 2>&1
result=$?
printf '%s\n' "$result" > "$out.exit"
date -u +%FT%TZ > "$out.ended"
tail -n 15 "$out.log"
exit "$result"
