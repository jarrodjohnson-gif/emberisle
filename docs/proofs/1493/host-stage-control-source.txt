import { spawn } from 'node:child_process';
import { mkdtempSync,rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import path from 'node:path';
const load=spawn(process.execPath,['--input-type=module','-e','setInterval(()=>{const until=performance.now()+100;while(performance.now()<until){}},200)'],{stdio:'ignore'});
try {
 for(let trial=1;trial<=5;trial++){
  const dir=mkdtempSync(path.join(tmpdir(),'emberisle-health-startup-'));
  const start=performance.now();let stdout='',stderr='';
  const host=spawn(process.execPath,['scripts/health-host-stage-diagnostic.mjs','--host-fixture'],{env:{...process.env,PORT:'0',ROOMS_DIR:dir,RESULTS_DIR:dir+'/results',PROFILES_DIR:dir+'/profiles',SAVE_MS:'20',TURN_MS:'120000',PING_MS:'60000'},stdio:['ignore','pipe','pipe','ipc']});
  host.stdout.on('data',d=>stdout+=d);host.stderr.on('data',d=>stderr+=d);
  const result=await new Promise(resolve=>{const timer=setTimeout(()=>resolve('timeout10000'),10000);host.stdout.on('data',()=>{if(/listening \d+/.test(stdout)){clearTimeout(timer);resolve('ready')}});host.once('exit',(code,signal)=>{clearTimeout(timer);resolve({code,signal})});});
  console.log(JSON.stringify({trial,result,elapsedMs:Math.round(performance.now()-start),stdout,stderr}));
  host.kill('SIGKILL');await new Promise(resolve=>host.exitCode!==null||host.signalCode!==null?resolve():host.once('exit',resolve));rmSync(dir,{recursive:true,force:true});
  if(result!=='ready'){process.exitCode=1;break;}
 }
}finally{load.kill()}
