// #748: the compact results list on touch and narrow screens, and the desktop table on a wide mouse screen.
// A finished four-player game with long names, buildings, both awards, hidden points, a tie and a zero total.
// Every row opens by tap and keyboard, one at a time; the open row scrolls to the top of the list and fits the
// list and the screen; every value matches the game state; nothing scrolls sideways; every result action stays reachable.
// The list is its own chunk: a real game fetches it while the table plays (no empty first paint), a table already over as
// it lands paints the dialog with its list too (#1695), a refused chunk leaves
// the table with every total until Show scores tries again, and the desktop never asks for it.
// With SHOTS=1 it writes the closed and one-open pictures in docs/proofs/748/ at device scale 2.
import assert from "node:assert/strict";
import { mkdirSync } from "node:fs";
import { chromium } from "./proof-browser.mjs";
import { createServer } from "vite";
import { createGame } from "../src/lib/game/board.ts";
import { publicScoreBreakdown, totalVP } from "../src/lib/game/rules.ts";

const port = Number(process.env.VITE_PORT || 18774);
const SHOTS = process.env.SHOTS === "1" ? "docs/proofs/748" : null;
const g = createGame({ humans: [{ name: "Bartholomew Featherstonehaugh" }, { name: "Guinevere Wigglesworth-Smythe" }, { name: "Maximilian Oppenheimer III" }, { name: "Cordelia Fitzgerald-Hartwell" }], bots: 0, seed: 559 });
g.rollOff = null; g.log = []; g.victoryTarget = 10;
const built = [["p0", "stronghold"], ["p0", "stronghold"], ["p0", "stronghold"], ["p0", "outpost"], ["p0", "outpost"], ["p1", "stronghold"], ["p1", "stronghold"], ["p1", "outpost"], ["p1", "outpost"], ["p2", "stronghold"], ["p2", "stronghold"]];
built.forEach(([playerId, kind], i) => { g.vertices[i].building = { playerId, kind }; });
g.longestRoad = "p0"; g.largestArmy = "p2"; g.players[2].knightsPlayed = 3;
g.players.forEach((p, i) => { p.hidden.vp = [1, 3, 3, 0][i]; });
g.phase = "over"; g.winner = "p0"; g.current = "p0";
g.winReason = { trigger: "outpost", points: totalVP(g, "p0"), target: 10 };

const pts = (n) => `${n} point${n === 1 ? "" : "s"}`;
const expected = Object.fromEntries(g.players.map((p) => {
  const parts = publicScoreBreakdown(g, p.id);
  const count = (kind) => g.vertices.filter((v) => v.building?.playerId === p.id && v.building.kind === kind).length;
  const total = totalVP(g, p.id);
  return [p.id, {
    name: p.name,
    total,
    lines: [
      ["Outposts", String(count("outpost")), pts(parts.outposts)],
      ["Strongholds", String(count("stronghold")), pts(parts.strongholds)],
      ["Longest path", parts.longestPath ? "Yes" : "No", pts(parts.longestPath)],
      ["Largest army", parts.largestArmy ? "Yes" : "No", pts(parts.largestArmy)],
      ["Hidden points", String(p.hidden.vp), pts(total - Object.values(parts).reduce((a, b) => a + b, 0))],
      ["Total", "", pts(total)],
    ],
  }];
}));
// The fixture covers what the issue asks for: a tie, a zero total, and zero parts next to non-zero ones.
assert.deepEqual(g.players.map((p) => expected[p.id].total), [11, 9, 9, 0]);
const order = ["p0", "p1", "p2", "p3"];

const LIST = [
  [440, 956, true],
  [956, 440, true],
  [390, 844, true],
  [430, 932, true],
  [820, 1180, true],
  [1180, 820, true],
  [844, 390, true],
  [667, 375, true, { top: 0, right: 44, bottom: 21, left: 44 }],
  [700, 900, false], // The phone rule's narrow half: a mouse under 768 px wide.
  [1280, 720, false], // Owner 2026-11-12 (q1563-results-desktop): the desktop shows the same list.
];

const vite = await createServer({ plugins: [{ name: "observe-win-wait-commit", enforce: "pre", transform(code, id) {
  if (!id.split("?")[0].endsWith("/src/components/game/WinScreen.tsx")) return;
  const target = "const timer = window.setTimeout(wait.out, LIST_WAIT_MS);";
  if (!code.includes(target)) throw new Error("WinScreen commit observation target missing");
  return code.replace(target, 'globalThis.__winTiming?.push({ kind: "win-commit", at: performance.now() }); ' + target);
}}], server: { host: "127.0.0.1", port, strictPort: true }, logLevel: "error" });
let browser, observedPage;
const errors = [];
const LIST_CHUNK = /\/WinList\.tsx/;
const SHEETS_CHUNK = /\/src\/components\/game\/sheets\.ts(\?|$)/;
try {
  await vite.listen();
  browser = await chromium.launch({ args: ["--use-gl=angle", "--use-angle=swiftshader", "--enable-unsafe-swiftshader"] });
  // #1701: the list chunk's request hangs, never answered and never failing. The dialog waits for it no longer than 4 s, then
  // shows with the plain list, every total, as a refused chunk does: never before 3.5 s after landing, always by 4.6 s. When
  // the held response is then let through, the list replaces the plain one.
  {
    const context = await browser.newContext({ viewport: { width: 390, height: 844 }, hasTouch: true, isMobile: true, reducedMotion: "reduce" });
    const page = await context.newPage();
    observedPage = page;
    await page.addInitScript(() => {
      for (const key of ["drawElements", "drawArrays", "drawElementsInstanced", "drawArraysInstanced"]) {
        const originalDraw = WebGL2RenderingContext.prototype[key];
        WebGL2RenderingContext.prototype[key] = function (...args) {
          if (!window.__controlledGlBlock && window.__landed && performance.now() - window.__landed >= 3800) {
            const started = performance.now(); window.__controlledGlBlock = { started };
            while (performance.now() - started < 2000) {}
            window.__controlledGlBlock.ended = performance.now();
            window.__winTiming.push({ kind: "controlled-GL-block", ...window.__controlledGlBlock });
          }
          return originalDraw.apply(this, args);
        };
      }
      // Controlled fixture only: same DOM/state/timers, no GL draw submissions.
      for (const key of ["drawElements", "drawArrays", "drawElementsInstanced", "drawArraysInstanced"]) {
        WebGL2RenderingContext.prototype[key] = function () {};
      }
      window.__winTiming = [];
      const record = (kind, extra = {}) => window.__winTiming.push({ kind, at: performance.now(), ...extra });
      const original = window.setTimeout;
      window.setTimeout = function (handler, delay, ...args) {
        if (delay !== 4000 || typeof handler !== "function") return original.call(this, handler, delay, ...args);
        record("timer-scheduled", { delay });
        return original.call(this, function (...values) { record("timer-fired"); return handler.apply(this, values); }, delay, ...args);
      };
      try { new PerformanceObserver(list => { for (const e of list.getEntries()) record("long-task", { start: e.startTime, duration: e.duration }); }).observe({ type: "longtask", buffered: true }); }
      catch (error) { record("long-task-unavailable", { message: error.message }); }
    });
    const tag = "390x844 landing, list chunk hangs";
    page.on("pageerror", (e) => errors.push(`${tag}: ${e.message}`));
    page.on("console", (m) => m.type() === "error" && errors.push(`${tag}: ${m.text()}`));
    const listLoads = [];
    const held = [];
    await page.route(LIST_CHUNK, (r) => void (listLoads.push(r.request().url()), held.push(r)));
    const sheetsAsked = page.waitForRequest(SHEETS_CHUNK);
    await page.goto(`http://127.0.0.1:${port}/`);
    await page.waitForFunction(() => !!window.__emberisle);
    await page.evaluate((url) => import(url).then(() => {}), (await sheetsAsked).url());
    await page.evaluate((game) => {
      window.__shown = null;
      new MutationObserver((_, o) => {
        const screen = document.querySelector('[data-testid="win-screen"]');
        if (!screen) return;
        window.__shown = { at: performance.now() - window.__landed, rows: [...screen.querySelectorAll('li[data-testid="win-row"]')].map((el) => [el.dataset.player, Number(el.querySelector('[data-testid="win-total"]').textContent)]), list: screen.querySelectorAll('button[data-testid="win-row"]').length };
        o.disconnect();
      }).observe(document.body, { childList: true, subtree: true });
      const s = window.__emberisle;
      s.setState({ spectator: false, isHost: true, menuFor: null, chatOpen: false });
      window.__landed = performance.now();
      s.getState().loadState(game, "p0", true, "WLST");
    }, structuredClone(g));
    await page.evaluate(() => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(() => r()))));
    assert.equal(await page.getByTestId("win-screen").count(), 0, `${tag}: no dialog in the first frames, it waits for the list`);
    await page.getByTestId("win-screen").waitFor({ timeout: 8000 });
    const shown = await page.evaluate(() => window.__shown);
    console.log("WIN_TIMING", JSON.stringify(await page.evaluate(() => ({ landed: window.__landed, shown: window.__shown, events: window.__winTiming, now: performance.now() }))));
    assert.ok(listLoads.length > 0, `${tag}: the list chunk was asked for`);
    assert.ok(shown.at >= 3500 && shown.at <= 4600, `${tag}: the dialog shows 3.5 to 4.6 s after landing (${Math.round(shown.at)} ms)`);
    assert.deepEqual(shown.rows, order.map((id) => [id, expected[id].total]), `${tag}: its first DOM update holds the plain list with every total`);
    assert.equal(shown.list, 0, `${tag}: no list while its chunk hangs`);
    // The chunk lands late after all: the list takes the plain list's place.
    for (const r of held) await r.continue();
    await page.locator('button[data-testid="win-row"]').first().waitFor();
    assert.equal(await page.locator('button[data-testid="win-row"]').count(), 4, `${tag}: a late chunk brings the list`);
    assert.equal(await page.locator('li[data-testid="win-row"]').count(), 0, `${tag}: and the plain list is gone`);
    await context.close();
    console.log(`ok ${tag}: no dialog while the list chunk hangs, then the plain list with every total at ${Math.round(shown.at)} ms; a late chunk then brings the list`);
  }


  assert.deepEqual(errors, [], "focused hanging-list case zero console/page errors");
  console.log("win timing focused case complete; other full cases unrun");
} catch (error) {
  let reportTimer;
  try { console.log("WIN_FAILURE_TIMING", JSON.stringify(await Promise.race([
    observedPage.evaluate(() => ({ landed: window.__landed, shown: window.__shown, events: window.__winTiming, now: performance.now() })),
    new Promise(resolve => { reportTimer = setTimeout(() => resolve({ reportTimedOut: true }), 3000); })
  ]))); } catch (reportError) { console.log("WIN_FAILURE_TIMING unavailable", reportError.message); }
  finally { clearTimeout(reportTimer); }
  throw error;
} finally { await browser?.close(); await vite.close(); }
